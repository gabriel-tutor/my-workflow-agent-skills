+const COUPONS: ReadonlyArray<{
+  code: string;
+  discountCents: (amounts: { subtotal: number; total: number }) => number;
+}> = [
+  { code: "SAVE10", discountCents: ({ total }) => Math.round((total * 10) / 100) },
+  { code: "FLAT5", discountCents: ({ subtotal }) => (subtotal >= 2000 ? 500 : 0) },
+];
+
+/** Total after the tier discount and the coupon, in integer cents. Throws on an unknown code. */
+export function applyCoupon(cart: Cart, code: string): number {
+  const coupon = COUPONS.find((c) => c.code === code);
+  if (!coupon) throw new Error(`Unknown coupon code: ${code}`);
+  const total = totalCents(cart);
+  return total - coupon.discountCents({ subtotal: subtotalCents(cart), total });
