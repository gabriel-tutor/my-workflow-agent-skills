+/**
+ * Total after the tier discount and the coupon `code`, in integer cents.
+ * SAVE10 takes 10% off the total; FLAT5 takes $5 off when the subtotal is at
+ * least $20 and leaves the total unchanged otherwise. Throws on an unknown code.
+ */
+export function applyCoupon(cart: Cart, code: string): number {
+  const total = totalCents(cart);
+  switch (code) {
+    case "SAVE10":
+      return total - Math.round((total * 10) / 100);
+    case "FLAT5":
+      return subtotalCents(cart) >= 2000 ? total - 500 : total;
+    default:
+      throw new Error(`unknown coupon code: ${code}`);
+  }
