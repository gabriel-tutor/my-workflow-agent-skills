+/** FLAT5 requires this much subtotal (before the tier discount), in integer cents. */
+const FLAT5_MIN_SUBTOTAL_CENTS = 2000;
+
+/**
+ * Total after the tier discount and then the coupon `code`, in integer cents.
+ * Codes are case-insensitive: SAVE10 takes 10% off the post-tier total; FLAT5 takes
+ * 500 cents off it once the subtotal reaches `FLAT5_MIN_SUBTOTAL_CENTS`.
+ * Throws on an unknown code or a coupon the cart does not qualify for.
+ */
+export function applyCoupon(cart: Cart, code: string): number {
+  const total = totalCents(cart);
+  switch (code.toUpperCase()) {
+    case "SAVE10":
+      return total - Math.round((total * 10) / 100);
+    case "FLAT5":
+      if (subtotalCents(cart) < FLAT5_MIN_SUBTOTAL_CENTS) {
+        throw new Error(`coupon FLAT5 not applicable: subtotal must be at least ${FLAT5_MIN_SUBTOTAL_CENTS} cents`);
+      }
+      return total - 500;
+    default:
+      throw new Error(`unknown coupon: ${code}`);
+  }
