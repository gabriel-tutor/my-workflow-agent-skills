+/**
+ * Total after the tier discount and then the coupon `code`, in integer cents.
+ * Throws for an unknown code, and for a flat coupon whose subtotal minimum
+ * (checked before the tier discount) is not met.
+ */
+export function applyCoupon(cart: Cart, code: string): number {
+  const normalized = code.toUpperCase();
+  const coupon = COUPONS.find((c) => c.code === normalized);
+  if (!coupon) throw new Error(`unknown coupon: ${code}`);
+  const total = totalCents(cart);
+  switch (coupon.kind) {
+    case "percent":
+      return total - Math.round((total * coupon.percent) / 100);
+    case "flat":
+      if (subtotalCents(cart) < coupon.minSubtotalCents) {
+        throw new Error(
+          `coupon ${coupon.code} not applicable: subtotal must be at least ${coupon.minSubtotalCents} cents`,
+        );
+      }
+      return total - coupon.cents;
+  }
